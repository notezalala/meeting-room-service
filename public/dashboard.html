const socket = io(); // LOAD socket.emit('get_all_requests'); socket.on('all_requests', (list) => { const container = document.getElementById('orders'); container.innerHTML = ""; // 🔥 HARD SORT (กัน DB เพี้ยน) list.sort((a, b) => b.id - a.id); list.forEach(addCard); }); // REALTIME socket.on('new_request', (data) => { if(document.getElementById(`card-${data.id}`)) return; addCard(data); }); // CARD function addCard(data){ const div = document.createElement('div'); div.className = "card"; div.id = `card-${data.id}`; div.innerHTML = `
${data.room || 'UNKNOWN'}
☕ Coffee : ${data.coffee}
💧 Water : ${data.water}
🍵 Tea : ${data.tea}

🕒 เวลาเสิร์ฟ : ${data.serve_time}

📝 หมายเหตุ : ${data.comment || '-'}
${data.status}
`; // 🔥 IMPORTANT: always top container.prepend(div); }
