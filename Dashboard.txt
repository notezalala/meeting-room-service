<div class="dashboard-container" id="orders"></div>

<script>

const socket = io();

const audio = new Audio("https://actions.google.com/sounds/v1/alarms/beep_short.ogg");

function playSound(){
    audio.currentTime = 0;
    audio.play().catch(()=>{});
}

// ================= STATUS =================
function translateStatus(status){
    if(status === 'PENDING') return 'รอดำเนินการ';
    if(status === 'pending') return 'รอดำเนินการ';
    if(status === 'preparing') return 'กำลังเตรียม';
    if(status === 'done') return 'เสร็จสิ้น';
    return status;
}

function statusClass(status){
    if(status === 'preparing') return 'status preparing';
    if(status === 'done') return 'status done';
    return 'status pending';
}

// ================= LOAD (สำคัญมาก) =================
socket.emit('get_all_requests');

socket.on('all_requests', (list) => {

    render(list); // ใช้ server data ตรงๆ
});

// ================= NEW =================
socket.on('new_request', (data) => {

    // 🔥 สำคัญ: ดึงใหม่ทั้งชุด ไม่ใช้ unshift
    socket.emit('get_all_requests');

    playSound();
});

// ================= REMOVE =================
socket.on('remove_request', () => {
    socket.emit('get_all_requests');
});

// ================= STATUS UPDATE =================
socket.on('status_updated', () => {
    socket.emit('get_all_requests');
});

// ================= RENDER =================
function render(list){

    const container = document.getElementById('orders');
    container.innerHTML = "";

    list.forEach((d) => {

        const div = document.createElement('div');
        div.className = 'card';
        div.id = `card-${d.id}`;

        div.innerHTML = `
            <div>

                <div class="room">
                    #${d.queue_number} ${d.room_name || '-'}
                    <br><br>
                </div>

                <div class="info">
                
                    👤 ผู้สั่ง: ${d.customer_name || '-'}<br><br>
                </div>
                <div class="info2">
                <img src="https://images.emojiterra.com/google/android-oreo/512px/2615.png"
                style="width:35px;height:40px;vertical-align:-3px;margin-right:5px;">
                Coffee : ${d.coffee}<br>
                <img src="https://png.pngtree.com/png-vector/20260104/ourmid/pngtree-a-glass-of-sparkling-water-on-white-background-png-image_18419981.webp"
                style="width:35px;height:40px;vertical-align:-3px;margin-right:5px;">
                Water : ${d.water}<br>
                <img src="https://em-content.zobj.net/source/microsoft-teams/337/teacup-without-handle_1f375.png"
                style="width:35px;height:40px;vertical-align:-3px;margin-right:5px;">                
                Tea : ${d.tea}<br><br>

                </div>

                </div>
                <div class="info3">

                    <br><br>
                    🕒 ต้องการเสิร์ฟ ${d.serve_time}<br>
                
                    📝 ${d.comment || '-'}

                </div>

                <div class="${statusClass(d.status)}" id="status-${d.id}">
                    ${translateStatus(d.status)}
                </div>

            </div>

            <div class="actions">
                <button class="btn-preparing" onclick="prep(${d.id})">
                    กำลังเตรียม
                </button>

                <button class="btn-done" onclick="done(${d.id})">
                    เสร็จสิ้น
                </button>
            </div>
        `;

        container.appendChild(div);
    });
}

// ================= ACTION =================
function prep(id){
    socket.emit('update_status',{id,status:'preparing'});
}

function done(id){
    socket.emit('update_status',{id,status:'done'});
}

</script>
